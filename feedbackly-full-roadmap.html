↗ Executive Summary

What We Are Building

A self-hosted visual feedback tool that lets your clients and QA teams report bugs directly from any website, with annotated screenshots and automatic technical context.

1

Embeddable Widget

Lightweight JavaScript snippet that clients embed on their websites. Captures screenshots, enables annotation, collects browser metadata, and submits feedback, all without leaving the page.

2

Team Dashboard

Next.js web app where your team views, triages, comments on, and resolves feedback items. Organised by project with filtering, search, and status management.

3

API & Integrations

RESTful API that receives feedback from the widget, stores data in Supabase, and can push issues to Jira, Trello, GitHub Issues, or any PM tool via webhooks.

Complete Database Schema

PostgreSQL tables with Row Level Security, managed through Supabase migrations.

-- ============================================
-- ORGANISATIONS & TEAMS
-- ============================================

CREATE TABLE organisations (
  id          uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  name        text NOT NULL,
  slug        text UNIQUE NOT NULL,
  plan        text DEFAULT 'free',        -- free | pro | enterprise
  created_at  timestamptz DEFAULT now(),
  updated_at  timestamptz DEFAULT now()
);

CREATE TABLE members (
  id              uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  organisation_id uuid REFERENCES organisations(id) ON DELETE CASCADE,
  user_id         uuid REFERENCES auth.users(id) ON DELETE CASCADE,
  role            text DEFAULT 'member',     -- owner | admin | member
  invited_at      timestamptz DEFAULT now(),
  accepted_at     timestamptz,
  UNIQUE(organisation_id, user_id)
);

-- ============================================
-- PROJECTS (one per client website)
-- ============================================

CREATE TABLE projects (
  id              uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  organisation_id uuid REFERENCES organisations(id) ON DELETE CASCADE,
  name            text NOT NULL,
  domain          text,                      -- e.g. "client-site.com"
  api_key         text UNIQUE DEFAULT 'proj_' || gen_random_uuid(),
  widget_config   jsonb DEFAULT '{
    "position": "bottom-right",
    "color": "#7C3AED",
    "collectConsole": true,
    "collectNetwork": false,
    "guestReporting": true
  }'::jsonb,
  is_active       boolean DEFAULT true,
  created_at      timestamptz DEFAULT now(),
  updated_at      timestamptz DEFAULT now()
);

-- ============================================
-- FEEDBACK (core entity)
-- ============================================

CREATE TYPE feedback_status AS ENUM (
  'open', 'in_progress', 'resolved', 'closed', 'wont_fix'
);

CREATE TYPE feedback_priority AS ENUM (
  'low', 'medium', 'high', 'critical'
);

CREATE TYPE feedback_type AS ENUM (
  'bug', 'suggestion', 'question', 'other'
);

CREATE TABLE feedback (
  id              uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  project_id      uuid REFERENCES projects(id) ON DELETE CASCADE,
  assigned_to     uuid REFERENCES auth.users(id),

  -- Reporter info (may be anonymous guest)
  reporter_name   text,
  reporter_email  text,

  -- Content
  title           text,
  description     text,
  type            feedback_type DEFAULT 'bug',
  status          feedback_status DEFAULT 'open',
  priority        feedback_priority DEFAULT 'medium',

  -- Screenshot
  screenshot_url  text,                      -- Supabase Storage path

  -- Technical metadata (auto-captured by widget)
  page_url        text NOT NULL,
  browser         text,                      -- "Chrome 120.0"
  os              text,                      -- "macOS 14.2"
  screen_size     text,                      -- "1920x1080"
  viewport_size   text,                      -- "1440x900"
  device_pixel_ratio numeric,
  user_agent      text,

  -- Console & network logs (JSON)
  console_logs    jsonb DEFAULT '[]'::jsonb,
  network_logs    jsonb DEFAULT '[]'::jsonb,

  -- Custom metadata (passed via widget SDK)
  custom_metadata jsonb DEFAULT '{}'::jsonb,

  -- External PM tool reference
  external_id     text,                      -- Jira issue key, GitHub issue #
  external_url    text,                      -- Direct link to the PM tool issue

  created_at      timestamptz DEFAULT now(),
  updated_at      timestamptz DEFAULT now()
);

-- Indexes for common queries
CREATE INDEX idx_feedback_project ON feedback(project_id);
CREATE INDEX idx_feedback_status ON feedback(status);
CREATE INDEX idx_feedback_created ON feedback(created_at DESC);
CREATE INDEX idx_feedback_assigned ON feedback(assigned_to);

-- ============================================
-- COMMENTS (threaded discussion per feedback)
-- ============================================

CREATE TABLE comments (
  id            uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  feedback_id   uuid REFERENCES feedback(id) ON DELETE CASCADE,
  user_id       uuid REFERENCES auth.users(id),
  body          text NOT NULL,
  is_internal   boolean DEFAULT false,   -- internal = team only, not sent to reporter
  created_at    timestamptz DEFAULT now()
);

-- ============================================
-- ATTACHMENTS (additional files per feedback)
-- ============================================

CREATE TABLE attachments (
  id            uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  feedback_id   uuid REFERENCES feedback(id) ON DELETE CASCADE,
  file_name     text NOT NULL,
  file_url      text NOT NULL,
  file_size     bigint,
  mime_type     text,
  created_at    timestamptz DEFAULT now()
);

-- ============================================
-- INTEGRATIONS (PM tool connections)
-- ============================================

CREATE TABLE integrations (
  id              uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  project_id      uuid REFERENCES projects(id) ON DELETE CASCADE,
  provider        text NOT NULL,           -- jira | trello | github | linear | slack
  config          jsonb NOT NULL,           -- provider-specific config (board ID, project key, etc.)
  access_token    text,                      -- encrypted OAuth token
  is_active       boolean DEFAULT true,
  auto_push       boolean DEFAULT false,     -- auto-create issue on new feedback?
  created_at      timestamptz DEFAULT now()
);

-- ============================================
-- ACTIVITY LOG (audit trail)
-- ============================================

CREATE TABLE activity_log (
  id            uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  feedback_id   uuid REFERENCES feedback(id) ON DELETE CASCADE,
  user_id       uuid REFERENCES auth.users(id),
  action        text NOT NULL,           -- status_changed | assigned | commented | pushed_to_jira
  details       jsonb,                     -- { "from": "open", "to": "resolved" }
  created_at    timestamptz DEFAULT now()
);

How the Widget Works

Step-by-step flow from script embed to feedback submission.

Step 1: Client Embeds the Script

<!-- Paste before </body> on any website -->
<script
  src="https://feedbackly.app/widget.js"
  data-project="proj_abc123def456"
  data-position="bottom-right"
  data-color="#7C3AED"
  data-guest="true"
></script>

Step 2: Widget Initialises (Shadow DOM)

// src/index.ts — Entry point
class FeedbacklyWidget {
  private shadowRoot: ShadowRoot;
  private config: WidgetConfig;

  constructor() {
    // Parse data-* attributes from the script tag
    this.config = parseConfig();

    // Create Shadow DOM container (isolates CSS from host site)
    const host = document.createElement('div');
    host.id = 'feedbackly-widget';
    document.body.appendChild(host);
    this.shadowRoot = host.attachShadow({ mode: 'closed' });

    // Inject widget styles into shadow DOM
    this.injectStyles();

    // Start console/network interceptors
    if (this.config.collectConsole) ConsoleCapture.start();
    if (this.config.collectNetwork) NetworkCapture.start();

    // Render floating action button
    this.renderButton();
  }
}

// Auto-initialise when script loads
new FeedbacklyWidget();

Step 3: Screenshot Capture + Annotation

// src/capture/screenshot.ts
import html2canvas from 'html2canvas';

export async function captureScreenshot(): Promise<string> {
  // Hide the widget before capturing
  widgetHost.style.display = 'none';

  const canvas = await html2canvas(document.body, {
    useCORS: true,
    scale: window.devicePixelRatio,
    logging: false,
    backgroundColor: null,
    ignoreElements: (el) => el.id === 'feedbackly-widget',
  });

  widgetHost.style.display = 'block';
  return canvas.toDataURL('image/png');
}

// src/annotation/canvas.ts
import { fabric } from 'fabric';

export function createAnnotationCanvas(screenshotDataUrl: string) {
  const canvas = new fabric.Canvas('annotation-canvas', {
    isDrawingMode: false,
    selection: true,
  });

  // Set screenshot as background
  fabric.Image.fromURL(screenshotDataUrl, (img) => {
    canvas.setBackgroundImage(img, canvas.renderAll.bind(canvas));
  });

  return canvas;
}

Step 4: Submit Feedback to API

// src/core/api.ts
export async function submitFeedback(data: FeedbackPayload) {
  // 1. Upload screenshot to Supabase Storage
  const screenshotBlob = dataURLtoBlob(data.screenshot);
  const fileName = `${data.projectKey}/${Date.now()}.png`;

  const { data: upload } = await supabase.storage
    .from('screenshots')
    .upload(fileName, screenshotBlob, { contentType: 'image/png' });

  // 2. Insert feedback record
  const { data: feedback } = await supabase
    .from('feedback')
    .insert({
      project_id: data.projectId,
      title: data.title,
      description: data.description,
      type: data.type,
      screenshot_url: upload.path,
      page_url: window.location.href,
      browser: getBrowserInfo(),
      os: getOSInfo(),
      screen_size: `${screen.width}x${screen.height}`,
      viewport_size: `${window.innerWidth}x${window.innerHeight}`,
      device_pixel_ratio: window.devicePixelRatio,
      user_agent: navigator.userAgent,
      console_logs: ConsoleCapture.getLogs(),
      network_logs: NetworkCapture.getLogs(),
      reporter_name: data.reporterName,
      reporter_email: data.reporterEmail,
    })
    .select()
    .single();

  return feedback;
}

↗ Quick Start

Getting Started Commands

Copy and run these commands to scaffold the entire project.

# 1. Create Turborepo monorepo
npx create-turbo@latest feedbackly
cd feedbackly

# 2. Set up Next.js app with Tailwind
cd apps/web
npx shadcn@latest init
npm install @supabase/supabase-js @supabase/ssr @tanstack/react-query zustand
npm install react-hook-form @hookform/resolvers zod
npm install resend date-fns sonner lucide-react

# 3. Set up Supabase locally
npx supabase init
npx supabase start   # starts local Postgres + Auth + Storage

# 4. Set up the widget package
cd ../../packages/widget
npm init -y
npm install html2canvas fabric @supabase/supabase-js
npm install -D vite typescript vite-plugin-css-injected-by-js

# 5. Create shared types package
cd ../shared
npm init -y
npm install -D typescript

# 6. Environment variables
cd ../../apps/web
cp .env.local.example .env.local
# Fill in:
#   NEXT_PUBLIC_SUPABASE_URL=
#   NEXT_PUBLIC_SUPABASE_ANON_KEY=
#   SUPABASE_SERVICE_ROLE_KEY=
#   RESEND_API_KEY=

# 7. Run development
cd ../..
npx turbo dev    # runs both web app and widget dev server
